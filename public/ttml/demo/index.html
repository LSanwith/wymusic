接口怎么调

// 1) 默认：本站自己的形状，TTML 内嵌成字符串
const r = await fetch('/search?ttml=186016');
const data = await r.json();
if (data.code === 200) console.log(data.ttml);
//   -> { code, id, ttml, url, size, uploadedAt }

// 2) 对标 Apple Music：返回 data[].attributes.ttmlLocalizations
//    /search?ttml=186016&format=apple
//    {
//      "data": [{
//        "id": "186016",
//        "type": "syllable-lyrics",
//        "attributes": {
//          "playParams": { "catalogId": "186016", "displayType": 3,
//                          "id": "AP_186016-zh", "kind": "lyric" },
//          "language": "zh",
//          "ttmlLocalizations": "<tt xmlns=..."
//        }
//      }]
//    }

// 3) 想要 TTML 文件本体（正确的 MIME），加 raw=1，会 302 到 Blob
//    /search?ttml=186016&raw=1
//    注意：application/ttml+xml 浏览器没有渲染器，会直接存盘

// 返回码
//   200  成功     400  ttml 参数不是纯数字
//   404  这个 id 还没有 TTML        501  服务端没配 Blob
